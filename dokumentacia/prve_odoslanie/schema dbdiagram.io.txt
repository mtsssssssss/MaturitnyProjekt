/* MATURITNY PROJEKT */

/********************************************************************/

// POUZIVATEL A JEHO TESTY
Table pouzivatel {
  id serial [pk]
  username varchar(32) [not null, unique]
  password varchar [not null]
  role enum("user", "admin") // admin bude moct pridavat otazky a predmety
  created_at timestamp 
}

Table test {
  id serial [pk]
  user_fk integer [not null]
  predmet_fk integer [not null]
  vytvoreny_datum timestamp
  dokonceny_datum datetime
  uspesnost double
}

// test by mal mat 30 otazok
Table test_otazka {
  id serial [pk]
  test_fk integer [not null]
  otazka_fk integer [not null]
  poradie integer [not null]
}

/********************************************************************/

// UKLADANIE ODPOVEDI ZADANYCH POUZIVATELOM
Table odpoved_pouzivatela {
  id serial [pk]
  test_otazka_fk integer [not null]
  zadana_odpoved varchar(256) // ak je otazka dopisovacia
  vybrana_moznost_fk integer // ak je otazka abcd
  je_spravna boolean // vyhodnotenie otazky 
}

Table odpoved_matching { // ak je otazka matching
  id serial [pk]
  odpoved_fk integer [not null]
  matching_otazka_moznosti_fk integer [not null]
  zadana_prava_strana varchar(256) [not null]
  je_spravna boolean 
}

/********************************************************************/

// PREDMETY A DATABAZA OTAZOK Z PREDMETOV
Table predmet {
  id serial [pk]
  subject_name varchar(64) [not null]
  subject_abbrev char(3) [not null, unique]
}

Table otazka {
  id serial [pk]
  subject_fk integer [not null]
  typ enum('abcd', 'dopisovacia', 'matching') [not null]
  text_otazky text [not null]
  spravna_odpoved varchar(256) // iba pri dopisovacich, pri abcd alebo matching je null
}

//max 4 moznosti
Table abcd_moznosti {
  id serial [pk]
  otazka_fk integer [not null]
  text varchar(256) [not null]
  je_spravna boolean 
}

//max 4 moznosti
Table matching_otazka_moznosti {
  id serial [pk]
  otazka_fk integer [not null]
  text_lava varchar(256) [not null]
  text_prava varchar(256) [not null]
}

/********************************************************************/

Ref: predmet.id < otazka.subject_fk
Ref: otazka.id < abcd_moznosti.otazka_fk
Ref: otazka.id < matching_otazka_moznosti.otazka_fk
Ref: pouzivatel.id < test.user_fk
Ref: predmet.id < test.predmet_fk
Ref: test.id < test_otazka.test_fk
Ref: otazka.id < test_otazka.otazka_fk
Ref: test_otazka.id < odpoved_pouzivatela.test_otazka_fk
Ref: abcd_moznosti.id < odpoved_pouzivatela.vybrana_moznost_fk
Ref: odpoved_pouzivatela.id < odpoved_matching.odpoved_fk
Ref: matching_otazka_moznosti.id < odpoved_matching.matching_otazka_moznosti_fk
